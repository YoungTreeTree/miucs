cmake_minimum_required(VERSION 3.24)
project(MiucsPlugins VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  add_compile_options(/utf-8)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
endif()
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../build/plugin-deps" CACHE PATH "Verified plugin dependencies")
file(GLOB LUA_SOURCES "${DEPS}/lua-5.4.8/src/*.c")
list(FILTER LUA_SOURCES EXCLUDE REGEX "/lua(c)?\\.c$")
add_library(lua STATIC ${LUA_SOURCES})
target_include_directories(lua PUBLIC "${DEPS}/lua-5.4.8/src")
add_library(plugin_core STATIC src/model.cpp src/resources.cpp src/runtime.cpp)
target_include_directories(plugin_core PUBLIC include src "${DEPS}")
target_link_libraries(plugin_core PUBLIC lua)
add_executable(miucs-plugin-worker src/worker.cpp)
target_link_libraries(miucs-plugin-worker PRIVATE plugin_core)
add_executable(plugin-tests ../../tests/plugins/core_tests.cpp)
target_link_libraries(plugin-tests PRIVATE plugin_core)
add_executable(plugin-benchmark ../../tests/plugins/benchmark.cpp)
target_link_libraries(plugin-benchmark PRIVATE plugin_core)
enable_testing()
add_test(NAME plugin-core COMMAND plugin-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../plugins/staff-note")
add_executable(plugin-exit-validation ../../tests/plugins/exit_validation.cpp)
target_link_libraries(plugin-exit-validation PRIVATE plugin_core)
add_test(NAME plugin-exit-validation COMMAND plugin-exit-validation)
set_tests_properties(plugin-exit-validation PROPERTIES TIMEOUT 10)
option(MIUCS_SKIA "Build Skia renderer and host bridge" ON)
if(MIUCS_SKIA)
  set(SKIA_ROOT "${DEPS}/skia-0746a1afaf750159e1a6aef29992980f4e293585" CACHE PATH "Pinned Skia headers")
  add_library(miucs_plugins SHARED src/client.cpp src/renderer.cpp src/bridge.cpp)
  target_include_directories(miucs_plugins PUBLIC include PRIVATE src "${DEPS}" "${SKIA_ROOT}")
  target_compile_definitions(miucs_plugins PRIVATE MIUCS_EXPORTS SK_GANESH)
  target_link_libraries(miucs_plugins PRIVATE plugin_core)
  if(APPLE)
    enable_language(OBJCXX)
    add_executable(plugin-surface-probe ../../tests/plugins/mac_surface.mm)
    target_compile_options(plugin-surface-probe PRIVATE -fobjc-arc)
    target_link_libraries(plugin-surface-probe PRIVATE miucs_plugins plugin_core "-framework AppKit" "-framework QuartzCore")
    target_sources(miucs_plugins PRIVATE src/surface_mac.mm)
    target_compile_definitions(miucs_plugins PRIVATE SK_METAL)
    target_compile_options(miucs_plugins PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    target_link_libraries(miucs_plugins PRIVATE "${DEPS}/skia-binaries/libskia.a" "-framework Metal" "-framework QuartzCore" "-framework AppKit" "-framework CoreText" "-framework CoreGraphics" "-framework ImageIO")
  elseif(WIN32)
    add_executable(plugin-surface-probe ../../tests/plugins/win_surface.cpp)
    target_link_libraries(plugin-surface-probe PRIVATE miucs_plugins plugin_core user32 ole32)
    target_sources(miucs_plugins PRIVATE src/surface_win.cpp)
    target_compile_definitions(miucs_plugins PRIVATE SK_DIRECT3D NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(miucs_plugins PRIVATE "${DEPS}/skia-binaries/skia.lib" d3d12 d3dcompiler dxgi dcomp dwrite windowscodecs user32 gdi32 ole32)
  endif()
endif()
